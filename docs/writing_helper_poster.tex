\begin{minipage}[t]{0.49\linewidth}
\PanelH{2.82in}{
  \Section{Motivation}
  \Body{Prompt-based personalization asks writers to describe style in advance. This project responds to a more ordinary writing situation: a writer stops, rejects, repairs, or asks for another version because the generated text is close but not right.}
  \vspace{0.05in}
  \DesignCard{Research question}{Can interruptions recover preferences about wording, tone, structure, and specificity?}
}
\end{minipage}
\hfill
\begin{minipage}[t]{0.49\linewidth}
\PanelH{2.82in}{
  \Section{Writing Preference Profile}
  \Body{The profile stores writing preferences: wording, specificity, rhythm, paragraph structure, transitions, and explanation style. It is not meant to infer personality, identity, ability, or private traits.}
  \vspace{0.07in}
  \DesignCard{Inclusion condition}{A preference enters the profile only after three similar interruption-and-repair observations, reducing one-click overinterpretation.}
}
\end{minipage}

\vspace{0.08in}

\begin{minipage}[t]{0.59\linewidth}
\PanelH{2.30in}{
  \Section{Process Highlights}
  {\fontsize{21}{24}\selectfont\color{Muted}
  \textbf{\color{Ink}Iteration 1:} I started with a generic writing helper, then made it interpret stopping points and generate replacements.\par
  \textbf{\color{Ink}Iteration 2:} I added an interpreter and rejected immediate profile updates, using a three-observation rule to reduce one-click overfitting.\par
  \textbf{\color{Ink}Iteration 3:} I generated random profiles and used an LLM to simulate the writing process, then summarized the data and recovery rate.\par}
}
\end{minipage}
\hfill
\begin{minipage}[t]{0.39\linewidth}
\PanelH{2.30in}{
  \Section{Reflection}
  {\fontsize{21}{24}\selectfont\color{Muted}
  \textbf{\color{Ink}Limit:} the simulation shows the mechanism works, but it does not prove real writers interrupt consistently.\par
  \textbf{\color{Ink}Next:} test with real writers and add controls to approve, edit, or delete inferred preferences.\par}
}
\end{minipage}
